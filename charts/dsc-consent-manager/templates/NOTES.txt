       --from-file={{ .Values.signingKeys.signingKeyFile }}=./signing-key.jwk
{{- if .Values.signingKeys.retiredVerificationKeys.enabled }} \
       --from-file={{ .Values.signingKeys.retiredVerificationKeysFile }}=./retired-verification-keys.jwks

     kubectl -n {{ .Release.Namespace }} rollout restart deployment/{{ $fullname }}
